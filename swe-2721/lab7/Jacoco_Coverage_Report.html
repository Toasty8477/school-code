  53.     public boolean addSynonymPair(String word, String synonym) throws WCException {
  54.         boolean retVal = false;

  57.         if (!determineIfStringisSpaceFree(word)) {

  67.         } else {

  73.             if (synonymList.isEmpty()) {
  74.                 synonymList.put(word, synonym);

  80.                     // Add the pair.
  81.                     synonymList.put(word, synonym);
  82.                     retVal = true;
  83.                 }
  84.             }
  85.         }

  100.     public String findSynonym(String word) throws WCException {
  101.         String retVal = null;

  103.         // If the word contains whitespace, throw an exception.
  104.         if (word.matches(".*\\s.*")) {
  105.             throw new WCException("The word can not contain whitespace.");
  106.         }

  108.         // Make sure the word does not have any non-alpha characters in it.
  109.         int i = 0;
  110.         do {    
  111.             if (Character.isLetter(word.charAt(i)) == false) {
  112.                 throw new WCException("Non-Alpha character found in the word.");
  113.             }
  114.             ++i;
  115.         } while (i < word.length());

  117.         // If the word is not a word, throw an exception.http://marketplace.eclipse.org/marketplace-client-intro?mpc_install=1163
  118.         if (word.length() < 2) {

  122.         // Now lets search for the word. Note that the following is not ideal by

  137.             }

  139.         return retVal;

  148.     protected String toLowerCase(String word) {
  149.         String retVal = null;
  150.         if (word != null) {
  151.             StringBuilder newWord = new StringBuilder(word);

  153.             int index = 0;
  154.             for (index = 0; index < newWord.length(); index++) {
  155.                 if (Character.isUpperCase(newWord.charAt(index))) {
  156.                     // We need to replace the character with a lower case version.
  157.                     newWord.setCharAt(index, Character.toLowerCase(newWord.charAt(index)));
  158.                 }
  159.             }
  160.             retVal = newWord.toString();
  161.         }
  162.         return retVal;
  163.     }

  171.     protected boolean determineIfStringisSpaceFree(String text) {
  172.         boolean retVal;
  173.         if (text == null) {
  174.             retVal = false;

  177.             retVal = true;

  181.             while (index >= 0 && retVal) {
  182.                 if (Character.isWhitespace(word.charAt(index))) {
  183.                     // We have found a whitespace within the string.
  184.                     retVal = false;
  185.                 }
  186.                 index--;
  187.             }
